Trigger Name: ReactivaCobro
Table: pago

BEGIN


      DECLARE contas INT DEFAULT 0;
    DECLARE totalPagado DECIMAL(15,2) DEFAULT 0;
    DECLARE totalFactura DECIMAL(15,2) DEFAULT 0;
    DECLARE totalPendiente DECIMAL(15,2) DEFAULT 0;

    SELECT 
        COALESCE(SUM(p.montocobrado), 0) AS totalPagado,
        fv.total AS totalFactura
    INTO 
        totalPagado, totalFactura
    FROM factura_venta fv
    LEFT JOIN pago p ON p.codNota = fv.codFacturaV
       AND p.ingresoegreso = 1 
       AND p.estado = 1 
       AND p.codTipoPago <> 12   
    WHERE fv.codFacturaV = NEW.codNota;

    SET contas = CASE 
                    WHEN totalPagado >= totalFactura THEN 1
                    ELSE 0
                 END;

    SET totalPendiente = totalFactura - totalPagado;

    UPDATE factura_venta 
    SET cancelado = contas,
        pendiente = totalPendiente,
        abonado = totalPagado
    WHERE codFacturaV = NEW.codNota;
END
